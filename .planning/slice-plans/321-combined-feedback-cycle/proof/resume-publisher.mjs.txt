import fs from 'node:fs';
import {execFileSync as exec} from 'node:child_process';
const B='/private/tmp/dough321-combined-proof/partial';
const S=JSON.parse(fs.readFileSync(B+'/partial-context.json'));
const inventory=fs.readFileSync(B+'/inventory.json','utf8');
const git=(r,...a)=>exec('git',['-C',r,...a],{encoding:'utf8'}).trim();
const save=()=>fs.writeFileSync(B+'/resume-context.json',JSON.stringify(S,null,2));
function recheck(){if(fs.readFileSync(B+'/inventory.json','utf8')!==inventory)throw Error('Inventory changed: rediscover through local owner');}
for(const p of Object.values(S)){
 if(git(p.localPath,'rev-parse','--show-toplevel')!==p.localPath||git(p.localPath,'remote','get-url','origin')!==p.origin)throw Error('Identity mismatch');
 git(p.localPath,'fetch','origin');
 git(p.localPath,'merge-base','--is-ancestor',p.finalAccepted,'origin/'+p.ref);
}
const alpha=S.alpha;
const privateNow=()=>Object.fromEntries(Object.entries({head:['rev-parse','HEAD'],status:['status','--porcelain'],index:['diff','--cached','--binary'],work:['diff','--binary']}).map(([k,a])=>[k,git(alpha.localPath,...a)]));
if(JSON.stringify(privateNow())!==JSON.stringify(alpha.initial))throw Error('Alpha private state differs');
if(git(alpha.workspace,'rev-parse','HEAD')!==alpha.sharedCandidate||git(alpha.workspace,'status','--porcelain'))throw Error('Alpha candidate differs');
async function publish(p,base,phase){
 const w=p.workspace,rt=w+'/.agents/skills/';
 const {publishExecutionIncrement}=await import(rt+'dough-execute-plan/scripts/execution-increment-publication.mjs');
 const {checkQueuedClosures}=await import(rt+'dough-land/scripts/queued-closure-check.mjs');
 git(w,'diff','--check'); const candidate=git(w,'rev-parse','HEAD'); save();
 p[phase]=await publishExecutionIncrement({workspace:w,branch:p.branch,previouslyPublishedBase:base,targetRef:'refs/heads/'+p.ref,remote:'origin',defaultCheckout:p.localPath,validatedCandidate:candidate,validate:async()=>{git(w,'diff','--check');return true;},onFetchedTarget:async()=>{const q=await checkQueuedClosures({checkout:w,remote:'origin',targetRef:'refs/heads/'+p.ref});return q.status==='clear'?undefined:{status:q.status,fields:q};},beforePush:async pair=>{recheck();fs.appendFileSync(B+'/resume-pushes.jsonl',JSON.stringify({project:p.id,phase,target:p.origin+' refs/heads/'+p.ref,...pair})+'\n');},register:()=>{}});
 save(); console.log(p.id,phase,JSON.stringify(p[phase]));
 if(!p[phase].ok)throw Error('Publication stopped; retained context has boundary');
 p.finalAccepted=p[phase].receipt.sha; save();
}
// Resume the retained shared owner through maintained landing, without recreating it.
await publish(alpha,alpha.local.receipt.sha,'resumedShared');
// Previously retired workspaces stay retired. Restored input needs fresh owned work.
for(const n of ['beta','open-dough']){
 const p=S[n];p.start=git(p.localPath,'rev-parse','origin/'+p.ref);
 p.workspace=p.localPath+'/.worktrees/feedback-cycle-resume';p.branch='maintenance/feedback-cycle-resume';p.created=true;
 git(p.localPath,'worktree','add','-b',p.branch,p.workspace,p.start);save();
}
const beta=S.beta,w=beta.workspace;
const raw=fs.readFileSync(w+'/DearDough.md','utf8');
const restoration=JSON.parse(fs.readFileSync(B+'/restoration.json'));
if(Buffer.from(raw).toString('hex')!==restoration.restoredSourceBytes||beta.start!==restoration.betaInputCommit)throw Error('Restored beta differs');
const local=fs.readFileSync(w+'/notes/ProjectFindings.md','utf8');
if(!local.includes('Execution beta-status')||!local.includes('25 minutes')||!fs.readFileSync(w+'/tools/status-label.mjs','utf8').includes('FAILED'))throw Error('Local preservation incomplete');
// Local phase: the restored DD-101 is the already-preserved execution, not new evidence.
fs.writeFileSync(w+'/DearDough.md',raw.slice(0,raw.indexOf('## DD-101'))+raw.slice(raw.indexOf('## DD-102')));
git(w,'diff','--check');git(w,'add','-A');git(w,'commit','-m','Remove restored duplicate local finding already preserved in canonical record');
beta.resumedLocalCandidate=git(w,'rev-parse','HEAD');save();
await publish(beta,beta.start,'resumedLocal');
recheck();
// Shared reconciliation retains the supplied beta execution at the existing owner.
const od=S['open-dough'], owner=od.workspace+'/docs/maintainer/findings/ODF-001.md';
const source=fs.readFileSync(w+'/DearDough.md');
const {createHash}=await import('node:crypto');const digest=createHash('sha256').update(source).digest('hex');
let body=fs.readFileSync(owner,'utf8');
body=body.replace('Three distinct project/execution occurrences; no inferred beta occurrence.',`- Project beta; execution beta-checks; 2026-10-17T10:00:00+09:00;
  Codex; untagged execution guidance G1, not the installed runtime version.
  Three redundant accepted-suite reruns cost 39 minutes with unchanged code and
  environment. Source: \`${beta.localPath}/DearDough.md\`, DD-102,
  post-local SHA ${beta.resumedLocal.receipt.sha}; source SHA256 ${digest}.
  Restored source input is recoverable at ${restoration.betaInputCommit}.
  Decisive evidence: \`${beta.localPath}/evidence/checks.txt\` and
  \`.open-dough/reported-guidance.md\` G1. Mapping: beta/DD-102 → ODF-001.

Four distinct project/execution occurrences (156 minutes total).`);
fs.writeFileSync(owner,body);git(od.workspace,'diff','--check');git(od.workspace,'add','-A');git(od.workspace,'commit','-m','Retain restored beta check-reuse occurrence under existing shared identity');
od.resumedSharedCandidate=git(od.workspace,'rev-parse','HEAD');save();
if(!fs.readFileSync(owner,'utf8').includes(digest)||!fs.readFileSync(w+'/DearDough.md').equals(source))throw Error('Evidence preservation failed');
fs.writeFileSync(w+'/DearDough.md',source.toString().replace('## DD-102 — Accepted suite repeated\n','## ODF-001 — Accepted suite repeated\nFormer local code: DD-102.\n'));
git(w,'diff','--check');git(w,'add','-A');git(w,'commit','-m','Adopt restored shared finding name with original evidence intact');
beta.resumedSharedCandidate=git(w,'rev-parse','HEAD');save();
await publish(od,od.start,'resumedShared');
await publish(beta,beta.resumedLocal.receipt.sha,'resumedShared');
for(const p of Object.values(S)){
 const rt=p.localPath+'/.agents/skills/';
 git(p.localPath,'fetch','origin');git(p.localPath,'merge-base','--is-ancestor',p.finalAccepted,'origin/'+p.ref);p.remoteTip=git(p.localPath,'rev-parse','origin/'+p.ref);
 const {refreshDefaultCheckout}=await import(rt+'dough-execute-plan/scripts/maintain-default-checkout.mjs');
 p.refresh=await refreshDefaultCheckout({checkout:p.localPath,integrationBranch:p.ref,remote:'origin'});save();
 if(fs.existsSync(p.workspace)){
  p.management=git(p.workspace,'rev-parse','--path-format=absolute','--git-common-dir');save();
  const {retireWorktree}=await import(rt+'dough-land/scripts/worktree-retirement.mjs');
  p.retirement=await retireWorktree({repository:p.management,worktree:p.workspace,branch:p.branch,remote:'origin',targetRef:'refs/heads/'+p.ref,createdForWork:true,contained:[p.finalAccepted]});save();
 }
 console.log(p.id,JSON.stringify({accepted:p.finalAccepted,refresh:p.refresh,retirement:p.retirement}));
}
if(JSON.stringify(privateNow())!==JSON.stringify(alpha.initial))throw Error('Private work changed');
try{git(alpha.localPath,'merge-base','--is-ancestor',alpha.initial.head,'origin/'+alpha.ref);throw Error('Private commit published');}catch(e){if(e.message==='Private commit published')throw e;}
alpha.unrelatedPreserved=true;recheck();save();
