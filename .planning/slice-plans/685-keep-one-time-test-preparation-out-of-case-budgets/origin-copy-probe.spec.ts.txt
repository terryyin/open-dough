import { existsSync } from "node:fs";
import path from "node:path";
import { expect, test } from "../../tests/support/pageTest.ts";
import { startOrigin, type StartOrigin } from "../../tests/support/startOrigin.ts";
import { keepStartedOrigin } from "../../tests/support/startOriginPreparation.ts";
import { queuedReadyRepository } from "../../../Distribution/tests/support/queued-repository.mjs";
import { precedeClaimWithPlanning } from "../../tests/support/storyCiComparisonFixture.ts";
import { completedPublishedStory, advanceCompletedTrunk } from "../../tests/support/completedPublishedStory.ts";
import { git, commitAll, writeAt } from "../../tests/support/storyReviewWorktree.ts";

test("real CLI completion history survives isolated copies and a peer write", async () => {
  const description = queuedReadyRepository().please();
  const source = await startOrigin(description);
  const copies: StartOrigin[] = [];
  try {
    await precedeClaimWithPlanning(source);
    const done = await completedPublishedStory(source, "story-branch");
    advanceCompletedTrunk(source);
    const head = git(source.origin, "rev-parse", "main");
    const refs = git(source.origin, "show-ref");
    const history = git(source.origin, "rev-list", "--all", "--parents");
    keepStartedOrigin("claude", description, source);
    for (let index = 0; index < 2; index += 1) {
      const copy = await startOrigin(description);
      copies.push(copy);
      expect(git(copy.origin, "show-ref")).toBe(refs);
      expect(git(copy.origin, "rev-list", "--all", "--parents")).toBe(history);
      expect(git(copy.project, "rev-parse", "HEAD")).toBe(head);
      expect(git(copy.project, "rev-parse", `${done.completion}^2`)).toBe(
        git(source.project, "rev-parse", `${done.completion}^2`),
      );
      expect(existsSync(path.join(copy.origin, "objects/info/alternates"))).toBe(false);
      expect(git(copy.project, "remote", "get-url", "origin")).toBe(copy.origin);
      git(copy.origin, "fsck", "--full");
    }
    writeAt(copies[0]!.project, "private.txt", "one test only\n");
    commitAll(copies[0]!.project, "private change");
    git(copies[0]!.project, "push", "--quiet", "origin", "main");
    expect(git(copies[1]!.origin, "rev-parse", "main")).toBe(head);
    expect(git(source.origin, "rev-parse", "main")).toBe(head);
  } finally {
    for (const copy of copies) copy.cleanup();
    source.cleanup();
  }
});
